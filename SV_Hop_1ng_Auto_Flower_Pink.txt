    -- // HTTP adapter: có pcall, kiểm tra status code, backoff khi 429/5xx

    local function httpGetJson(url, onNote)
        local remain = CONFIG.MIN_FETCH_INTERVAL - (os.clock() - lastFetchTime)
        if remain > 0 then
            task.wait(remain)
        end
        lastFetchTime = os.clock()

        if httpRequest then
            local attempt = 0
            while attempt <= CONFIG.RETRY_MAX do
                local ok, res = pcall(httpRequest, {
                    Url = url,
                    Method = "GET",
                    Headers = { ["Accept"] = "application/json" },
                })
                if not ok or type(res) ~= "table" then
                    return nil, "Executor không gửi được request HTTP"
                end
                local code = tonumber(res.StatusCode) or tonumber(res.Status) or 0
                if code == 200 then
                    if type(res.Body) == "string" and res.Body ~= "" then
                        local okJson, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
                        if okJson and type(data) == "table" then
                            return data, nil
                        end
                        return nil, "Dữ liệu trả về không phải JSON hợp lệ"
                    end
                    return nil, "Phản hồi rỗng từ Roblox"
                elseif code == 429 then
                    if attempt >= CONFIG.RETRY_MAX then
                        return nil, "Bị Roblox giới hạn request (429) — thử lại sau ít phút"
                    end
                    local backoff = CONFIG.RETRY_BASE * (attempt + 1)
                    if onNote then
                        onNote(("⚠ Bị giới hạn 429 — chờ %ds rồi thử lại…"):format(backoff), THEME.WAIT)
                    end
                    task.wait(backoff)
                    attempt = attempt + 1
                elseif code >= 500 then
                    if attempt >= CONFIG.RETRY_MAX then
                        return nil, "Lỗi máy chủ Roblox (HTTP " .. tostring(code) .. ")"
                    end
                    task.wait(CONFIG.RETRY_BASE * (attempt + 1))
                    attempt = attempt + 1
                else
                    return nil, "Lỗi HTTP " .. tostring(code)
                end
            end
            return nil, "Không lấy được danh sách server"
        end

        -- Dự phòng khi executor không có request()
        local okGet, raw = pcall(function()
            return game:HttpGet(url)
        end)
        if not okGet or type(raw) ~= "string" or raw == "" then
            return nil, "Executor không hỗ trợ HTTP — không lấy được danh sách server"
        end
        local okJson, data = pcall(HttpService.JSONDecode, HttpService, raw)
        if not okJson or type(data) ~= "table" then
            return nil, "Dữ liệu trả về không phải JSON hợp lệ"
        end
        return data, nil
    end

    -- // Quét danh sách server (nhiều trang, có cache)
    local cache = { candidates = nil, threshold = nil, scanned = 0, at = 0 }

    local function scanServers(threshold, onNote)
        if cache.candidates and cache.threshold == threshold
            and (os.clock() - cache.at) < CONFIG.CACHE_TIME then
            return cache.candidates, nil, cache.scanned
        end

        local candidates = {}
        local scanned = 0
        local cursor = nil

        for page = 1, CONFIG.MAX_PAGES_PER_SCAN do
            local url = ("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):format(PlaceId)
            if cursor then
                url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
            end
            if onNote then
                onNote(("Đang quét danh sách server (trang %d/%d)…"):format(page, CONFIG.MAX_PAGES_PER_SCAN), THEME.WAIT)
            end

            local data, err = httpGetJson(url, onNote)
            if not data then
                if #candidates > 0 then break end
                return nil, err, 0
            end

            for _, s in ipairs(data.data or {}) do
                scanned = scanned + 1
                local sid = s.id and tostring(s.id) or nil
                if sid and sid ~= tostring(game.JobId) then
                    local playing = s.playing or 0
                    local maxPlayers = s.maxPlayers or 0
                    local notFull = (maxPlayers == 0) or (playing < maxPlayers)
                    if notFull and playing == threshold then
                        table.insert(candidates, {
                            id = sid,
                            playing = playing,
                            maxPlayers = maxPlayers,
                            ping = s.ping,
                        })
                    end
                end
            end

            if #candidates > 0 then
                break -- đã có ứng viên đạt yêu cầu, khỏi quét thêm
            end

            cursor = data.nextPageCursor
            if not cursor or cursor == "" then break end
            task.wait(CONFIG.PAGE_DELAY)
        end

        -- Ít người nhất trước, ping thấp hơn làm phụ
        table.sort(candidates, function(a, b)
            if a.playing ~= b.playing then
                return a.playing < b.playing
            end
            return (a.ping or 999) < (b.ping or 999)
        end)

        cache.candidates = candidates
        cache.threshold = threshold
        cache.scanned = scanned
        cache.at = os.clock()

        return candidates, nil, scanned
    end

    -- // Ngưỡng cố định: từ 3 người trở lên thì tìm server có đúng 1 người.
    -- Sau khi vào server đó, tổng số người sẽ là 2 (em + 1 người khác).
    local function getThreshold(_silent)
        return 1 -- số người mục tiêu trên server đích, không tính người chơi của mình
    end

    -- // Hop 1 lần
    local function hopOnce(fromAuto)
        if hopInProgress then return false end

        local targetPlayers = getThreshold(not fromAuto)
        if not targetPlayers then return false end

        local remain = CONFIG.HOP_COOLDOWN - (os.clock() - lastHopAttempt)
        if remain > 0 then
            if fromAuto then
                task.wait(remain)
            else
                setStatus(("Vừa hop xong — chờ khoảng %.0fs nữa nhé."):format(remain), THEME.WAIT)
                return false
end
        end

        -- Chỉ quét API server khi lần kiểm tra định kỳ thấy có >= 3 người.
        -- 1 hoặc 2 người: không gọi API, không teleport.
        local count = playerCount()
        if count < 3 then
            setStatus(("✓ Server có %d người — giữ nguyên; chỉ hop khi có từ 3 người."):format(count), THEME.OK)
            return false
        end

        local scanWait = nextScanAllowedAt - os.clock()
        if scanWait > 0 then
            return false
        end

        hopInProgress = true
        lastHopAttempt = os.clock()
        setStatus(("Server có %d người — đang tìm server có đúng 1 người…"):format(count), THEME.WAIT)

        local candidates, err, scanned = scanServers(targetPlayers, function(msg, color)
            setStatus(msg, color)
        end)

        if not candidates then
            hopInProgress = false
            nextScanAllowedAt = os.clock() + CONFIG.NO_RESULT_BACKOFF
            setStatus("Lỗi lấy danh sách server: " .. tostring(err), THEME.ERR)
            return false
        end

        if #candidates == 0 then
            hopInProgress = false
            nextScanAllowedAt = os.clock() + CONFIG.NO_RESULT_BACKOFF
            setStatus(("Chưa tìm thấy server có đúng 1 người sau khi quét %d server. Sẽ thử lại theo chu kỳ."):format(scanned), THEME.WAIT)
            return false
        end

        local best = candidates[1]
        setStatus(("🎯 Chọn server tốt nhất: %d/%d người (trong %d server đã quét) — đang hop…"):format(best.playing, best.maxPlayers, scanned), THEME.ACCENT)

        local ok, terr = pcall(function()
            TeleportService:TeleportToPlaceInstance(PlaceId, best.id, LocalPlayer)
        end)

        if ok then
            setStatus(("Đang chuyển tới server %d/%d người… Chúc may mắn! 🌸"):format(best.playing, best.maxPlayers), THEME.OK)
            return true
        else
            hopInProgress = false
            setStatus("Teleport lỗi: " .. tostring(terr):sub(1, 80), THEME.ERR)
            return false
        end
    end

    -- // Báo lỗi teleport
    TeleportService.TeleportInitFailed:Connect(function(plr, result, message)
        if plr == LocalPlayer then
            hopInProgress = false
            setStatus("Teleport thất bại (" .. tostring(result.Name) .. ") — có thể thử hop lại.", THEME.ERR)
        end
    end)

    -- // Nút Auto Hop (có cờ hủy, bật/tắt nhiều lần không sinh vòng lặp trùng)
    updateAutoBtn = function()
        if autoEnabled then
            AutoBtn.Text = "⏹ Auto Hop: ĐANG BẬT"
            AutoBtn.BackgroundColor3 = THEME.BTN_AUTO_ON
        else
            AutoBtn.Text = "▶ Auto Hop: TẮT"
            AutoBtn.BackgroundColor3 = THEME.BTN_AUTO_OFF
        end
    end
    updateAutoBtn()

    -- Không có nút bật/tắt: Auto Hop luôn tự khởi chạy.
    local function startAutoHop()
        autoEnabled = true
        autoGen = autoGen + 1
        local gen = autoGen
        task.spawn(function()
            while autoEnabled and gen == autoGen and running and ScreenGui.Parent do
                hopOnce(true)
                -- Một lần kiểm tra mỗi 60 giây; không quét danh sách server liên tục.
                task.wait(CONFIG.AUTO_CHECK_INTERVAL)
            end
        end)
    end

    -- // Nút Hop ngay
    HopBtn.MouseButton1Click:Connect(function()
        task.spawn(hopOnce, false)
    end)

    -- // Nút quét lại
    RefreshBtn.MouseButton1Click:Connect(function()
        cache.candidates = nil
        cache.at = 0
        setStatus("Đã xóa cache — lần hop tới sẽ quét danh sách mới.", THEME.TEXT_SUB)
    end)

    -- // Thu nhỏ / Đóng
    local minimized = false
    MinBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        Content.Visible = not minimized
        MainFrame.Size = minimized and CLOSED_SIZE or OPEN_SIZE
        MinBtn.Text = minimized and "+" or "–"
    end)

    -- Hộp xác nhận lần hai khi nhấn X.
    local ConfirmOverlay = make("Frame", {
        Name = "CloseConfirmOverlay",
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundColor3 = Color3.fromRGB(70, 20, 45),
        BackgroundTransparency = 0.25,
        BorderSizePixel = 0,
        Visible = false,
        ZIndex = 50,
    }, MainFrame)
    corner(16, ConfirmOverlay)

    local ConfirmCard = make("Frame", {
        Size = UDim2.new(1, -36, 0, 150),
        Position = UDim2.new(0, 18, 0.5, -75),
        BackgroundColor3 = THEME.CARD,
        BorderSizePixel = 0,
        ZIndex = 51,
    }, ConfirmOverlay)
    corner(14, ConfirmCard)
    stroke(THEME.HEADER_1, 2, ConfirmCard)

    make("TextLabel", {
        Size = UDim2.new(1, -20, 0, 34),
        Position = UDim2.new(0, 10, 0, 12),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        Text = "🌸 Xác nhận đóng UI?",
        TextColor3 = THEME.TEXT_MAIN,
        TextSize = 17,
        ZIndex = 52,
    }, ConfirmCard)

    make("TextLabel", {
        Size = UDim2.new(1, -24, 0, 36),
        Position = UDim2.new(0, 12, 0, 47),
        BackgroundTransparency = 1,
        Font = Enum.Font.Gotham,
        Text = "Bạn có thể mở lại bằng nút hoa 🌸.",
        TextColor3 = THEME.TEXT_SUB,
        TextSize = 12,
        TextWrapped = true,
        ZIndex = 52,
    }, ConfirmCard)

    local ConfirmYes = make("TextButton", {
        Size = UDim2.new(0.5, -18, 0, 34),
        Position = UDim2.new(0, 12, 1, -46),
        BackgroundColor3 = THEME.HEADER_2,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamBold,
        Text = "Đóng",
        TextColor3 = THEME.WHITE,
        TextSize = 13,
        ZIndex = 52,
    }, ConfirmCard)
    corner(9, ConfirmYes)

    local ConfirmNo = make("TextButton", {
        Size = UDim2.new(0.5, -18, 0, 34),
        Position = UDim2.new(0.5, 6, 1, -46),
        BackgroundColor3 = Color3.fromRGB(255, 190, 220),
        BorderSizePixel = 0,
        Font = Enum.Font.GothamBold,
        Text = "Hủy",
        TextColor3 = THEME.TEXT_MAIN,
        TextSize = 13,
        ZIndex = 52,
    }, ConfirmCard)
    corner(9, ConfirmNo)

    CloseBtn.MouseButton1Click:Connect(function()
        ConfirmOverlay.Visible = true
    end)

    ConfirmNo.MouseButton1Click:Connect(function()
        ConfirmOverlay.Visible = false
    end)

    ConfirmYes.MouseButton1Click:Connect(function()
        ConfirmOverlay.Visible = false
        -- Xác nhận lần hai: ẩn cửa sổ chính, giữ nút hoa để mở lại.
        autoEnabled = false
        autoGen = autoGen + 1
        updateAutoBtn()
        MainFrame.Visible = false
    end)

    -- // Vòng đếm người realtime (có điều kiện dừng)
    task.spawn(function()
        while running and ScreenGui.Parent do
            refreshNow()
            task.wait(CONFIG.AUTO_CHECK_INTERVAL)
        end
    end)

    -- // Khởi động
    refreshNow()
    setStatus("Chạy nền: kiểm tra mỗi 60 giây; từ 3 người trở lên mới tìm server 1 người.", THEME.ACCENT)
    -- Tắt toàn bộ ScreenGui: không bảng trạng thái, không nút hoa, không menu.
    ScreenGui.Enabled = false
    startAutoHop()
end
